Быстрый старт

Ниже показан минимальный путь, который соответствует текущему коду Botify.

Регистрация в DI

using Botify.Builder;
using Microsoft.Extensions.Hosting;

var builder = Host.CreateApplicationBuilder(args);

var token = builder.Configuration["Telegram:BotToken"]
    ?? throw new InvalidOperationException("Telegram:BotToken не задан");

builder.Services.AddBotifyHandlers();
builder.Services.AddBotify(options =>
{
    options.SetToken(token);
});

var host = builder.Build();
host.Run();

Минимальный handler

using Botify;
using Botify.Attributes;
using Microsoft.Extensions.DependencyInjection;
using Telegram.Bot;

[CommandHandler]
public class StartHandler
{
    [Command("start", "Начать работу с ботом")]
    public async Task Start(BotifyContext context)
    {
        if (context.Chat is null)
            return;

        await context.Client.SendTextMessageAsync(
            context.Chat.Id,
            "Привет! Botify готов к работе.",
            cancellationToken: context.CancellationToken);
    }
}

Что важно помнить

AddBotifyHandlers() сканирует загруженные сборки и ищет:

  • классы с атрибутами [CommandHandler], [CallbackHandler], [MessageHandler], [InlineHandler], [PaymentHandler];
  • типы, реализующие IBotifyValidator.

Методы внутри handler-классов должны быть public, нестатическими и иметь сигнатуру Task Method(BotifyContext context).
